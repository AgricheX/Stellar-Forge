namespace = painted_galaxy_parent

country_event = {
	id = painted_galaxy_parent.1
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		# not all countries have been created yet, so we need parents to check for children and vice versa
		# note: assume there's always 1 parent; that's the player's resposibility
		if = {
			limit = { painted_galaxy_is_parent = yes }
			save_event_target_as = painted_galaxy_parent
			every_country = {
				limit = { prev = { painted_galaxy_is_parent_of = { CHILD = prev } } }
				save_event_target_as = painted_galaxy_child
				painted_galaxy_setup_parent_child_flags = yes
			}
		}
		if = {
			limit = { painted_galaxy_is_child = yes }
			save_event_target_as = painted_galaxy_child
			random_country = {
				limit = { painted_galaxy_is_parent_of = { CHILD = prev } }
				save_event_target_as = painted_galaxy_parent
				painted_galaxy_setup_parent_child_flags = yes
			}
		}
	}
}
